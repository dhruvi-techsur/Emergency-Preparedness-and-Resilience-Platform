```typescript
import type { PoolClient } from 'pg';

export class AiOriginNotApprovedError extends Error {}

/**
 * The ONE enforcement point for every endpoint writing ai_origin:true
 * content to an official record (ics_forms here; notifications/sitreps/aars
 * in a future Phase 4). Per FRD F9 Validation: "No API endpoint that writes
 * to an incident, ICS form, notification, SITREP, or AAR may accept content
 * flagged ai_origin: true without a non-null approved_by/approved_at pair —
 * this is validated at the data layer, independent of which UI client calls
 * it." Always call this INSIDE the same transaction as the domain write it
 * guards, using the same `client`.
 */
export async function assertAiOriginApproved(
  client: PoolClient,
  payload: { ai_origin?: boolean; ai_draft_id?: string },
): Promise<void> {
  if (!payload.ai_origin) return;
  if (!payload.ai_draft_id) {
    throw new AiOriginNotApprovedError('ai_origin set with no ai_draft_id');
  }
  const result = await client.query<{ approved_by: string | null }>(
    `SELECT approved_by FROM ai_drafts WHERE id = $1`,
    [payload.ai_draft_id],
  );
  if (!result.rows[0] || result.rows[0].approved_by === null) {
    throw new AiOriginNotApprovedError('ai_draft not approved');
  }
}
```
