-- Same documented deviation 01-01 used for audit_log: DEPLOY-01's single-
-- DATABASE_URL deployment gives the app exactly one DB role, which must also
-- run migrations, so a separate low-privilege "app_runtime" role (TechArch's
-- literal `REVOKE UPDATE, DELETE ON incident_timeline FROM app_runtime`) is not
-- available. A BEFORE UPDATE/DELETE trigger gives the identical "enforced at
-- the data layer regardless of caller" guarantee.
CREATE OR REPLACE FUNCTION reject_incident_timeline_mutation() RETURNS trigger AS $$
BEGIN
  RAISE EXCEPTION 'incident_timeline is append-only: % not permitted', TG_OP;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_incident_timeline_immutable
  BEFORE UPDATE OR DELETE ON incident_timeline
  FOR EACH ROW EXECUTE FUNCTION reject_incident_timeline_mutation();

CREATE INDEX idx_incidents_jurisdiction_status ON incidents(jurisdiction_id, status);
CREATE INDEX idx_incidents_severity ON incidents(severity) WHERE status = 'Open';
CREATE INDEX idx_incidents_location ON incidents USING GIST(location);
CREATE INDEX idx_incident_command_roles_incident ON incident_command_roles(incident_id) WHERE unassigned_at IS NULL;
CREATE INDEX idx_incident_timeline_incident_time ON incident_timeline(incident_id, occurred_at DESC);

```typescript
import { PoolClient } from 'pg';

export type TimelineTone = 'brand' | 'success' | 'warning' | 'danger';

// This is the ONLY code path permitted to write to incident_timeline — mirrors
// auth-org's AuditService.record() single-writer pattern. ALWAYS call this
// inside the same transaction (same `client`) as the domain mutation it
// documents, never as a separate connection/call that could partially fail.
export const TimelineService = {
  async append(
    client: PoolClient,
    incidentId: string,
    title: string,
    meta: string | undefined,
    tone: TimelineTone,
    actorId: string | null,
    source: string,
    detail?: Record<string, unknown>
  ): Promise<void> {
    await client.query(
      `INSERT INTO incident_timeline (incident_id, title, meta, tone, actor_id, source, detail)
       VALUES ($1,$2,$3,$4,$5,$6,$7)`,
      [incidentId, title, meta ?? null, tone, actorId, source, detail ? JSON.stringify(detail) : null]
    );
  },
};

// Shared transaction helper every incident-mutating handler uses: runs `fn`
// inside BEGIN/COMMIT, rolling back (and rethrowing as TIMELINE_WRITE_FAILED
// upstream) if either the domain mutation or the TimelineService.append call
// fails — this is what makes "every state-changing action writes exactly one
// Timeline entry" a real transactional guarantee, not just a code convention.
import { pool } from '../../db/client';
export async function withTimelineTransaction<T>(fn: (client: PoolClient) => Promise<T>): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    const result = await fn(client);
    await client.query('COMMIT');
    return result;
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```
