export async function connectNats(): Promise<void> {
  nc = await connect({ servers: process.env.NATS_URL ?? 'nats://localhost:4222' });
  const jsm = await nc.jetstreamManager();
  await jsm.streams.add({ name: 'EPMP_EVENTS', subjects: ['epmp.events.>'] }).catch(() => {});

export async function publishEvent(eventType: string, payload: object): Promise<void> {
  if (!nc) throw new Error('NATS not connected');
  const js = nc.jetstream();
  await js.publish(`epmp.events.${eventType}`, codec.encode({ event_type: eventType, payload, published_at: new Date().toISOString() }));
}
export function getNatsConnection(): NatsConnection | undefined { return nc; }

Create `backend/src/modules/integration-gateway/webhook-delivery-worker.ts`: a JetStream **pull consumer** subscribed to `epmp.events.>`, started from `server.ts` after `connectNats()` succeeds (skip starting the worker if NATS connection failed, matching the graceful-degradation principle above).
```typescript
export async function startWebhookDeliveryWorker(): Promise<void> {
  const nc = getNatsConnection();
  if (!nc) return;
  const js = nc.jetstream();
  const sub = await js.pullSubscribe('epmp.events.>', { config: { durable_name: 'webhook-delivery' } });
  (async () => {
    for await (const msg of sub) {
      const decoded = codec.decode(msg.data) as { event_type: string; payload: object };
      const subs = await query(`SELECT * FROM webhook_subscriptions WHERE active=true AND $1 = ANY(event_types)`, [decoded.event_type]);
      for (const subscription of subs) {
        const idempotencyKey = crypto.createHash('sha256').update(`${subscription.id}:${msg.seq}`).digest('hex');
        const inserted = await query(
          `INSERT INTO webhook_deliveries (subscription_id, idempotency_key, event_type, payload)
           VALUES ($1,$2,$3,$4) ON CONFLICT (subscription_id, idempotency_key) DO NOTHING RETURNING id`,
          [subscription.id, idempotencyKey, decoded.event_type, JSON.stringify(decoded.payload)]
        );
        if (inserted.length === 0) continue; // DUPLICATE_DELIVERY_IGNORED — already processed, silent no-op per Y2-errors.md
        try {
          const res = await fetch(subscription.callback_url, {
            method: 'POST',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ idempotency_key: idempotencyKey, event_type: decoded.event_type, payload: decoded.payload, delivered_at: new Date().toISOString() }),
          });
          await query(`UPDATE webhook_deliveries SET delivery_status=$1, attempt_count=attempt_count+1 WHERE subscription_id=$2 AND idempotency_key=$3`,
            [res.ok ? 'delivered' : 'failed', subscription.id, idempotencyKey]);
        } catch {
          await query(`UPDATE webhook_deliveries SET delivery_status='failed', attempt_count=attempt_count+1, next_retry_at=now() + interval '1 minute' WHERE subscription_id=$1 AND idempotency_key=$2`,
            [subscription.id, idempotencyKey]);
        }
      }
      msg.ack();
    }
  })();
}
```
Also upsert a `background_jobs` row (`job_type='webhook_delivery'`) on worker start (`status='running'`, `last_run_at=now()`), updated on each processing loop iteration (`failure_count` incremented on delivery failure, `next_retry_at` set per the exponential-backoff convention from `Y2-errors.md`: base interval doubling per consecutive failure, capped at 1 hour — implement as a small helper `computeBackoff(failureCount: number): Date` used both here and anywhere else background_jobs needs it).
